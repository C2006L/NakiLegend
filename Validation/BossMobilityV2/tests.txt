  p.woodlandMode=false;Chapter();Enter(6);Quiet();Set("clock",50f);Set("lastDamageAt",45f);Set("shield",1f);Call("TickPlayerDefenses",1f);Check(F("shield")==1,"During a live fight five quiet seconds no longer restore the shield");Set("clock",51.6f);Call("TickPlayerDefenses",1f);Check(Mathf.Abs(F("shield")-1.55f)<.001f,"Combat shield recovery is exactly .55 per second after 6.5 seconds");KillCurrent();Quiet();Set("clock",60f);Set("shield",1f);Set("lastDamageAt",54f);Call("TickPlayerDefenses",1f);Check(F("shield")==2,"Safe cleared room retains one shield per second recovery");

 static void AuditBossMotion(int island,int stage,List<string> report){
  Quiet();Set("position",new Vector2(0,-4));Set("invulnerableUntil",100000f);object boss=List("enemies")[0];bool rooted=(bool)Call("RootedBoss",boss);
  for(int phase=0;phase<3;phase++){
   Set(boss,"hp",(float)Get(boss,"maximum")*(phase==0?1:phase==1?.65f:.3f));float distance=0;int attacks=0;
   for(int i=0;i<260;i++){Set("clock",F("clock")+.05f);Vector2 old=(Vector2)Get(boss,"position");float tell=(float)Get(boss,"windup");Call("UpdateEnemy",boss,.05f);if(tell<=0&&(float)Get(boss,"windup")>0)attacks++;distance+=Vector2.Distance(old,(Vector2)Get(boss,"position"));if(i%20==0)ClearShots();}
   report.Add(island+","+stage+","+Get(boss,"kind")+","+phase+","+distance.ToString("0.00")+","+attacks+","+rooted);
   File.WriteAllLines(Out+"boss-mobility.csv",report);
   Check(rooted||distance>.35f,"Actual boss AI has pursuit or repositioning in phase "+phase+": "+island+" / "+stage+" / "+Get(boss,"kind")+" moved "+distance);
   Check(attacks>=1,"Repositioning still allows warned attacks: "+island+" / "+stage+" / "+phase);
  }
 }
