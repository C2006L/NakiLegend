 static void MotionProbe(string kind,int island){
  Enter(island==0?Find("Combat"):island==1?1:island==2?6:island==3?31:32);ClearActors();List("obstacles").Clear();List("projectileObstacles").Clear();object enemy=Spawn(kind,Vector2.zero);Set("clock",500f);Set("invulnerableUntil",100000f);Set(enemy,"nextAttack",501f);Set(enemy,"attackDirection",Vector2.right);Set(enemy,"windup",.6f);Set(enemy,"recoveryUntil",0f);Set(enemy,"actionStarted",500f);Set(enemy,"attackNumber",1);Set(enemy,"hurtUntil",0f);((SpriteRenderer)Get(enemy,"body")).flipX=false;
  int flickers=0;float largest=0;for(int i=0;i<240;i++){
   Vector2 before=(Vector2)Get(enemy,"position");bool locked=(float)Get(enemy,"windup")>0||(float)Get(enemy,"chargeUntil")>F("clock")||F("clock")<(float)Get(enemy,"recoveryUntil");Vector2 direction=(Vector2)Get(enemy,"attackDirection");Set("position",before+new Vector2(i%2==0?-.02f:.02f,2));Set("clock",500+(i+1)*.025f);Call("UpdateEnemy",enemy,.025f);Vector2 after=(Vector2)Get(enemy,"position");largest=Mathf.Max(largest,Vector2.Distance(before,after));
   if(locked&&direction.x>.5f&&((SpriteRenderer)Get(enemy,"body")).flipX)flickers++;
  }
  Check(flickers==0,"Attack facing stays locked through centreline crossings: "+island+" / "+kind);Check(largest<.45f,"No frame-to-frame navigation teleport during natural attack cycle: "+island+" / "+kind+" max="+largest.ToString("F3"));
 }
