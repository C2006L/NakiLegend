 static void ProbeQuayExit(int direction,int island){
  Vector2 n=(Vector2)Call("DoorNormal",direction),t=new Vector2(-n.y,n.x),door=(Vector2)Call("DoorPoint",direction,0f);
  foreach(float edge in new[]{-.65f,0f,.65f}){
   Vector2 at=door-n*1.2f+t*edge;Check(!(bool)Call("Blocked",at,.3f),"Harbor door has full-width dry inside approach: "+island+" / "+direction+" / "+edge);
   Vector2 end=at;for(int k=0;k<45;k++)end=(Vector2)Call("MoveInExploration",end,n*.06f,.3f);
   Set("position",end);Set("doorCooldownUntil",0f);Set("doorTransition",0f);Set("clock",F("clock")+1);Call("TryAutomaticExit");
   Check(F("doorTransition")>0,"Walking through an off-centre harbor exit actually starts transition: "+island+" / "+direction+" / "+edge);Set("doorTransition",0f);
  }
 }
