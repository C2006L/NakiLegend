 static void RunPressureProbes(){
  var rows=new List<string>{"scenario,seconds,boss_health_left,player_health,shield,hits,kills,casts"};
  foreach(int profile in new[]{0,1,2}){
   bool immortal=profile>0,light=profile==2;StartCore(0,!light);Enter(42);Quiet();Set("origin",EnumVal("Origin",light?"Light":"Dark"));Set("energy",light?100f:0f);Set("darkPower",light?0f:300f);Set("elementPower",100f);Set("primarySkill",0);Set("orbLearned",true);Set("position",new Vector2(0,-4));Set("health",6f);Set("shield",F("EffectiveMaximumShield"));Set("invulnerableUntil",immortal?100000f:0f);Set("clock",10f);Set("nextSpell",0f);Set("nextSword",0f);((int[])Get("carriedWeapons"))[0]=3;
   object boss=List("enemies")[0];Set("encounterRecord",null);Call("BeginEncounterRecord");float began=F("clock");int hitCount=0,casts=0,kills=0;
   for(int step=0;step<(light?3600:1800)&&(float)Get(boss,"hp")>0&&F("health")>0;step++){
    float time=F("clock")+.05f;Set("clock",time);Call("TickPlayerDefenses",.05f);Call("TickEssence",.05f);if(light)Set("energy",Mathf.Min(100,F("energy")+.3f));
    if(time>=F("nextSword"))Call("UseEquippedWeapon");if(time>=F("nextSpell"))Call(light?"CastLightVolley":"CastDarkMeteor");
    foreach(object e in new ArrayList(List("enemies")))if(List("enemies").Contains(e))Call("UpdateEnemy",e,.05f);
    for(int i=List("shots").Count-1;i>=0;i--)if(i<List("shots").Count)Call("UpdateShot",i,.05f);
    object record=Get("encounterRecord");if(record!=null){hitCount=(int)Get(record,"hits");casts=(int)Get(record,"casts");kills=(int)Get(record,"kills");}
    Call("TickJourney",.05f);
   }
   rows.Add((light?"light_stationary_upper_damage_probe":immortal?"dark_stationary_upper_damage_probe":"dark_stationary_no_dodge")+","+(F("clock")-began).ToString("0.00")+","+Mathf.Max(0,(float)Get(boss,"hp")).ToString("0.0")+","+F("health").ToString("0.0")+","+F("shield").ToString("0.0")+","+hitCount+","+kills+","+casts);
   Check(immortal?(float)Get(boss,"hp")<=0:F("health")<=0||hitCount>=3,"Sustained real combat probe exercises the boss, projectile damage and resource loop: "+immortal);
  }
  File.WriteAllLines(Out+"combat-probes.csv",rows);
 }
