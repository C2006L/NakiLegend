 static bool CoverRoomConnected()
 {
  Rect bounds=(Rect)Get("CurrentRoomBounds");const float step=.3f;int width=Mathf.FloorToInt((bounds.width-.8f)/step)+1,height=Mathf.FloorToInt((bounds.height-.8f)/step)+1;var valid=new bool[width,height];var reached=new bool[width,height];var queue=new Queue<Vector2Int>();float closest=float.MaxValue;Vector2Int first=Vector2Int.zero;Vector2 player=(Vector2)Get("position");
  for(int y=0;y<height;y++)for(int x=0;x<width;x++){Vector2 at=new Vector2(bounds.xMin+.4f+x*step,bounds.yMin+.4f+y*step);valid[x,y]=!(bool)Call("Blocked",at,.34f);float distance=(at-player).sqrMagnitude;if(valid[x,y]&&distance<closest){closest=distance;first=new Vector2Int(x,y);}}
  if(closest==float.MaxValue)return false;queue.Enqueue(first);reached[first.x,first.y]=true;var directions=new[]{Vector2Int.up,Vector2Int.down,Vector2Int.left,Vector2Int.right};
  while(queue.Count>0){Vector2Int at=queue.Dequeue();foreach(Vector2Int direction in directions){Vector2Int next=at+direction;if(next.x<0||next.y<0||next.x>=width||next.y>=height||!valid[next.x,next.y]||reached[next.x,next.y])continue;reached[next.x,next.y]=true;queue.Enqueue(next);}}
  // A closed rescue cage intentionally has an inaccessible interior. Check the
  // actual exits and both sides of every usable cover, not that quest interior.
  var goals=new List<Vector2>();for(int d=0;d<4;d++)if((bool)Call("HasDoor",d))goals.Add((Vector2)Call("SafeArrivalPoint",d));
  foreach(var sr in ((Transform)Get("world")).GetComponentsInChildren<SpriteRenderer>())if(sr.gameObject.activeInHierarchy&&sr.gameObject.name=="Woodland natural cover"){Bounds body=sr.bounds;goals.Add(new Vector2(body.min.x-.55f,body.center.y));goals.Add(new Vector2(body.max.x+.55f,body.center.y));goals.Add(new Vector2(body.center.x,body.min.y-.55f));goals.Add(new Vector2(body.center.x,body.max.y+.55f));}
  foreach(Vector2 goal in goals){bool found=false;for(int y=0;y<height&&!found;y++)for(int x=0;x<width;x++)if(reached[x,y]&&Vector2.Distance(goal,new Vector2(bounds.xMin+.4f+x*step,bounds.yMin+.4f+y*step))<.42f){found=true;break;}if(!found)return false;}
  return true;
 }
